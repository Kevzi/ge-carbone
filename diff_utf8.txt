+    """

+    def patch(self, request, pk, entry_id):
+        user = request.user
+        if not user.cabinet:
+            return Response({'error': 'No cabinet'}, status=status.HTTP_403_FORBIDDEN)
+            
+        report = get_object_or_404(Report, id=pk, cabinet=user.cabinet)
+        entry = get_object_or_404(CarbonEntry, id=entry_id, report=report)
+        
+        serializer = CarbonEntryUpdateSerializer(data=request.data)
+        if not serializer.is_valid():
+            return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)
+            
+        validated_data = serializer.validated_data
+        physical_quantity = validated_data['physical_quantity']
+        physical_unit = validated_data['physical_unit']
+        emission_factor_id = validated_data['emission_factor_id']
+        
+        # Ensure emission factor exists
+        new_factor = get_object_or_404(EmissionFactor, id=emission_factor_id)
+        
+        if not new_factor.value_kg_co2_per_unit:
+            return Response({
+                'error': 'Le facteur d\'├®mission s├®lectionn├® n\'a pas de valeur par unit├® physique.'
+            }, status=status.HTTP_400_BAD_REQUEST)
+        
+        old_co2_kg = entry.co2_kg
+        old_factor_name = entry.emission_factor.name if entry.emission_factor else 'N/A'
+        old_factor_unit = entry.emission_factor.unit if entry.emission_factor else 'N/A'
+        
+        # Update entry
+        entry.physical_quantity = physical_quantity
+        entry.physical_unit = physical_unit
+        entry.emission_factor = new_factor
+        entry.mapping_method = 'manual'
+        entry.dqr = 1  # Expert confirmation: explicit physical value = DQR 1
+        
+        # Calculate new CO2
+        entry.co2_kg = physical_quantity * new_factor.value_kg_co2_per_unit
+        entry.scope = new_factor.scope
+        entry.save()

+        # Recalculate Report Totals
+        from django.db.models import Sum
+        totals = CarbonEntry.objects.filter(report=report).aggregate(
+            total=Sum('co2_kg'),
+            scope1=Sum('co2_kg', filter=models.Q(scope=1)),
+            scope2=Sum('co2_kg', filter=models.Q(scope=2)),
+            scope3=Sum('co2_kg', filter=models.Q(scope=3))
+        )
+        
+        report.total_co2_kg = totals['total'] or 0
+        report.scope1_co2_kg = totals['scope1'] or 0
